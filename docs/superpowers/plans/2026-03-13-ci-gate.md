# CI/CD Quality Gate Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `barad-dur check` subcommand that runs analysis and exits non-zero if the overall score drops below a configurable threshold or regresses vs. the previous history entry. Designed for CI pipelines.

**Architecture:** New `Check(CheckArgs)` variant in `Commands` enum. Reuses the existing analysis pipeline (collect + metrics + scoring) via an extracted helper function, then compares the result against thresholds. Exit codes: 0 = pass, 1 = threshold violation, 2 = regression detected.

**Tech Stack:** Rust, clap derive API, existing `scorer::AnalysisReport` and `cache::history` modules.

```rust
fn run_analysis_pipeline(pipe: PipelineArgs) -> Result<PipelineResult> {
    // Resolve target: URL -> clone, or local path
    let _temp_clone: Option<remote::clone::TempClone>;
    let (local_path, remote_meta): (PathBuf, Option<RemoteMeta>) = if remote::is_url(&pipe.target) {
        let clone = remote::clone::clone_remote(&pipe.target)?;
        let gh_meta = pipe.token.as_deref().and_then(|t| {
            if remote::github::is_github_url(&pipe.target) {
                match remote::github::fetch_meta(&pipe.target, t) {
                    Ok(m) => Some(m),
                    Err(e) => { eprintln!("Warning: GitHub API error: {}", e); None }
                }
            } else { None }
        });
        let path = clone.path.clone();
        _temp_clone = Some(clone);
        let meta = gh_meta.map(|m| RemoteMeta {
            url: pipe.target.clone(),
            stars: Some(m.stars),
            description: m.description,
            language: m.language,
            open_issues: Some(m.open_issues),
        });
        (path, meta)
    } else {
        _temp_clone = None;
        (PathBuf::from(&pipe.target), None)
    };

    // Load and merge config
    let cfg = config::load(&local_path)?;
    // Build a temporary AnalyzeArgs-compatible view for merge_with_cli
    let merge_args = AnalyzeArgs {
        target: pipe.target.clone(),
        token: pipe.token.clone(),
        health: false, team: false, evolution: false, hygiene: false,
        since: pipe.since.clone(),
        until: pipe.until.clone(),
        all: pipe.all,
        json: false, html: false, open: false, pretty: false,
        output: None, verbose: pipe.verbose,
        exclude: pipe.exclude.clone(),
        no_default_excludes: pipe.no_default_excludes,
        skip_blame: pipe.skip_blame,
        no_cache: pipe.no_cache,
        cache_only: pipe.cache_only,
    };
    let cfg = config::merge_with_cli(cfg, &merge_args);
    config::validate(&cfg)?;

    let time_window = build_time_window_from_config(&cfg, &merge_args);
    let collector = Collector::open(&local_path, time_window)?;

    if collector.is_shallow() {
        eprintln!("Warning: This is a shallow clone. Metrics may be incomplete.");
    }

    let show_progress = std::io::stderr().is_terminal();
    let current_head = collector.head_commit_hash()?;
    let exclude_patterns = &cfg.exclude_patterns;
    let use_default_excludes = cfg.exclude_use_defaults;

    let snapshot = if pipe.no_cache {
        collect_and_cache(&collector, show_progress, pipe.verbose > 0, cfg.skip_blame, true, exclude_patterns, use_default_excludes)?
    } else if let Some(cached) = cache::load(collector.repo_path())? {
        if !cache::is_stale(&cached, &current_head) {
            if pipe.verbose > 0 { eprintln!("Using cached snapshot."); }
            cached
        } else {
            if pipe.verbose > 0 { eprintln!("Cache stale, re-collecting..."); }
            collect_and_cache(&collector, show_progress, pipe.verbose > 0, cfg.skip_blame, false, exclude_patterns, use_default_excludes)?
        }
    } else if pipe.cache_only {
        bail!("No cache found. Run without --cache-only first.");
    } else {
        collect_and_cache(&collector, show_progress, pipe.verbose > 0, cfg.skip_blame, false, exclude_patterns, use_default_excludes)?
    };

    if snapshot.commits.is_empty() {
        eprintln!("Warning: No commits found in the specified time window.");
    }

    // Compute all metrics (check always runs all categories)
    let t = std::time::Instant::now();
    let categories = compute_all_metrics(&snapshot, &cfg);
    if pipe.verbose > 0 {
        eprintln!("  Metrics: {}ms", t.elapsed().as_millis());
    }

    // Score
    let t = std::time::Instant::now();
    let weight_pairs = cfg.weights.as_weight_pairs();
    let mut report = scorer::build_report(&snapshot, categories, remote_meta, &weight_pairs);
    if pipe.verbose > 0 {
        eprintln!("  Scoring: {}ms", t.elapsed().as_millis());
    }

    // Record history entry
    let history_entry = scorer::build_history_entry(&report, &current_head);
    if let Err(e) = cache::history::append_if_new_head(&history_entry, &local_path) {
        eprintln!("Warning: Failed to record history: {}", e);
    }
    report.history = cache::history::load_history(&local_path).unwrap_or_default();

    Ok(PipelineResult { report, local_path, cfg, _temp_clone })
}
```

```rust
/// Outcome of the `check` subcommand.
enum CheckOutcome {
    Pass,
    ThresholdViolation { score: u32, min_score: u32 },
    Regression { current: u32, previous: u32 },
}

impl CheckOutcome {
    fn exit_code(&self) -> i32 {
        match self {
            CheckOutcome::Pass => 0,
            CheckOutcome::ThresholdViolation { .. } => 1,
            CheckOutcome::Regression { .. } => 2,
        }
    }
}
```

- [ ] **Step 2: Implement `run_check()`**

```rust
fn run_check(args: CheckArgs) -> Result<CheckOutcome> {
    let min_score = args.min_score;
    let no_regression = args.no_regression;
    let output_format = args.output_format.clone();
    let verbose = args.verbose;

    let pipe_args = PipelineArgs::from(&args);
    let result = run_analysis_pipeline(pipe_args)?;
    let report = &result.report;

    // Check 1: threshold
    let mut outcome = if report.overall_score < min_score {
        CheckOutcome::ThresholdViolation {
            score: report.overall_score,
            min_score,
        }
    } else {
        CheckOutcome::Pass
    };

    // Check 2: regression (only if --no-regression and threshold passed or not)
    // Regression is a separate, stronger signal — report it even if threshold also failed.
    if no_regression {
        let history = &report.history;
        // Find the previous entry (second-to-last, since we just appended the current one)
        if history.len() >= 2 {
            let previous = &history[history.len() - 2];
            if report.overall_score < previous.overall_score {
                // Regression takes precedence over threshold violation
                outcome = CheckOutcome::Regression {
                    current: report.overall_score,
                    previous: previous.overall_score,
                };
            }
        }
    }

    // Output
    match output_format {
        CheckOutputFormat::Summary => {
            print_check_summary(report, &outcome, verbose);
        }
        CheckOutputFormat::Json => {
            let json = serde_json::to_string_pretty(report)?;
            println!("{}", json);
        }
    }

    Ok(outcome)
}

fn print_check_summary(
    report: &scorer::AnalysisReport,
    outcome: &CheckOutcome,
    verbose: u8,
) {
    eprintln!("barad-dur check: {}", report.repo_name);
    eprintln!("  Overall score: {}/100", report.overall_score);

    if verbose > 0 {
        for cat in &report.categories {
            eprintln!("  {}: {}/100", cat.name, cat.score);
        }
    }

    match outcome {
        CheckOutcome::Pass => {
            eprintln!("  Result: PASS");
        }
        CheckOutcome::ThresholdViolation { score, min_score } => {
            eprintln!(
                "  Result: FAIL — score {} is below minimum {}",
                score, min_score
            );
        }
        CheckOutcome::Regression { current, previous } => {
            eprintln!(
                "  Result: FAIL — regression detected: {} -> {} (was {})",
                previous, current, previous
            );
        }
    }
}
```
